% #1 x, #2 y, #3 heading deg, #4 field of view deg, #5 range, #6 style
\newcommand{\fov}[6]{%
  \path[#6] (#1,#2) -- ++({#3-#4/2}:#5)
    arc[start angle={#3-#4/2},delta angle=#4,radius=#5] -- cycle;}
% #1 x, #2 y, #3 length in m, #4 tick label step in m
\newcommand{\scalebar}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \fill[navy] (0,0) rectangle (#3/2,1.2pt);
    \draw[navy,line width=0.4pt] (0,0) rectangle (#3,1.2pt);
    \foreach \t in {0,#4,...,#3} {\node[below,inner sep=1pt,font=\sffamily\tiny] at (\t,0) {\t};}
    \node[right,font=\sffamily\tiny] at (#3,0.6pt) {m};
  \end{scope}}
\newcommand{\northarrow}[2]{%
  \begin{scope}[shift={(#1,#2)}]
    \fill[navy] (0,0) -- (-2.2pt,-1.5pt) -- (0,7pt) -- cycle;
    \draw[navy,line width=0.4pt] (0,0) -- (2.2pt,-1.5pt) -- (0,7pt);
    \node[above,inner sep=1pt,font=\sffamily\tiny\bfseries] at (0,7pt) {N};
  \end{scope}}
